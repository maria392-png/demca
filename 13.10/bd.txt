create table roles(
id serial primary key,
name varchar
);

create table users(
id serial primary key,
surname varchar,
name varchar,
lastname varchar,
login varchar,
id_role int references roles(id)
);

create table categories(
id serial primary key,
name varchar
);

create table subcategories(
id serial primary key,
name varchar
);

create table manufactures(
id serial primary key,
name varchar
);

create table products(
id serial primary key,
name varchar,
id_category int references categories(id),
id_subcategory int references subcategories(id),
pin varchar,
id_manufacture int references manufactures(id),
opisanie varchar,
sostav varchar,
price float
);

create table sizes(
id serial primary key,
name varchar
);

create table stocks(
id serial primary key,
id_product int references products(id),
id_size int references sizes(id),
count int
);

create table orders(
id serial primary key,
id_user int references users(id),
date date
);

create table orders_items(
id serial primary key,
id_order int references orders(id),
id_stock int references stocks(id),
count int
);
